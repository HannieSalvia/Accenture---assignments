=========================
UPDATE OPERATIONS
=========================

1)
db.products.updateOne(
{name:"iPhone 14"},
{$set:{name:"iPhone 15"}}
)

Output:
Product name updated successfully


2)
db.products.updateMany(
{price:{$lt:5000}},
{$set:{stock:100}}
)

Output:
Stock updated for products below 5000


3)
db.products.updateOne(
{brand:"Dell"},
{$set:{price:60000}}
)

Output:
Dell Laptop price updated


4)
db.products.updateMany(
{category:"Electronics"},
{$set:{stock:70}}
)

Output:
Stock updated for electronics products


5)
db.products.replaceOne(
{name:"Nike Shoes"},
{
name:"Puma Shoes",
category:"Fashion",
price:5500,
stock:60,
brand:"Puma"
}
)

Output:
Nike Shoes document replaced
